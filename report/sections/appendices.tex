\section{Symbols}
\begin{list}{}{\setlength{\labelwidth}{.2\columnwidth}\setlength{\leftmargin}{.24\columnwidth}\setlength{\itemsep}{0pt}\setlength{\parsep}{0pt}\renewcommand{\makelabel}[1]{#1\hfil}}
\item[$\mathbf{a}_k^{(e)}$] sum of training vectors assigned to source unit $k$ in epoch $e$
\item[$\alpha$] prevalence-correction strength
\item[$b$] best-matching-unit index function
\item[$c$] target-class index
\item[$d$] contextual quantisation-error function
\item[$e$] batch-update epoch index
\item[$f_c$] false positives for class $c$
\item[$F_{1,c}$] precision--recall score for class $c$
\item[$F_{1,\mathrm{macro}}$] mean class score across all 10 classes
\item[$g$] number of target classes
\item[$h_{jk}^{(e)}$] Gaussian influence from source $k$ to destination $j$ in epoch $e$
\item[$j$] destination map-unit index
\item[$k$] source map-unit index
\item[$K$] number of codebook clusters
\item[$\ell$] map side length
\item[$m$] number of map units
\item[$\boldsymbol{\mu}_T$] training feature centroid
\item[$n_c$] total training count of class $c$
\item[$n_k^{(e)}$] training count assigned to source unit $k$ in epoch $e$
\item[$n_X$] number of evaluated observations in $X$
\item[$n_{zc}$] training count assigned to cluster $z$ and class $c$
\item[$p_c$] precision for class $c$
\item[$q$] quantisation-error function
\item[$r_c$] recall for class $c$
\item[$\mathbf{r}_j$] lattice coordinate of unit $j$
\item[$s_{zc}$] prevalence-correction score function for cluster $z$ and class $c$
\item[$\sigma_e$] neighbourhood radius in epoch $e$
\item[$t_c$] true positives for class $c$
\item[$T$] training partition
\item[$u_c$] false negatives for class $c$
\item[$v$] non-negative numerical feature value
\item[$\mathbf{w}_j$] codebook vector of unit $j$
\item[$\mathbf{x}$] transformed observation vector
\item[$X$] evaluated partition
\item[$z$] codebook-cluster index
\end{list}
